## 代码整合

为了方便你在本地运行和测试，我们采用标准的 ES Modules（ESM）规范。我们规划了清晰的单向依赖结构（DAG），完全避开了模块循环引用的陷阱：

### 1. 📄 `effect.js`（核心基座）
负责管理 `activeEffect`、全局依赖图 `targetMap`，以及实现依赖的收集（`track`）与触发（`trigger`）。

```javascript
// 全局指针，用于指向当前正在运行的副作用函数
export let activeEffect = null;

// 存储所有对象属性依赖的全局图表 (key: target -> value: depsMap)
export const targetMap = new WeakMap();

/**
 * 声明副作用函数
 */
export function effect(fn, options = {}) {
  const effectFn = () => {
    let lastEffect = activeEffect;
    try {
      cleanupEffect(effectFn); // 每次执行前清理旧的依赖
      activeEffect = effectFn; // 标记当前活跃 effect
      return fn();             // 运行用户函数，并返回其结果
    } finally {
      activeEffect = lastEffect; // 恢复上一个活跃 effect（支持嵌套）
    }
  };

  effectFn.deps = [];
  effectFn.options = options;
  effectFn.scheduler = options.scheduler;

  // 惰性执行：如果配置了 lazy，不默认执行
  if (!options.lazy) {
    effectFn();
  }
  
  return effectFn;
}

/**
 * 清理依赖集合
 */
function cleanupEffect(effectFn) {
  const { deps } = effectFn;
  if (deps.length) {
    for (let i = 0; i < deps.length; i++) {
      deps[i].delete(effectFn);
    }
    deps.length = 0;
  }
}

/**
 * 在 get 阶段开始追踪依赖
 */
export function track(target, key) {
  if (!activeEffect) return;

  let depsMap = targetMap.get(target);
  if (!depsMap) {
    depsMap = new Map();
    targetMap.set(target, depsMap);
  }

  let dep = depsMap.get(key);
  if (!dep) {
    dep = new Set();
    depsMap.set(key, dep);
  }

  trackEffects(dep);
}

/**
 * 双向收集具体的依赖
 */
export function trackEffects(dep) {
  if (!dep.has(activeEffect)) {
    dep.add(activeEffect);
    activeEffect.deps.push(dep); // 反向收集
  }
}

/**
 * 在 set 阶段触发更新
 */
export function trigger(target, key) {
  const depsMap = targetMap.get(target);
  if (!depsMap) return;

  const dep = depsMap.get(key);
  if (!dep) return;

  triggerEffects(dep);
}

/**
 * 遍历并触发更新
 */
export function triggerEffects(dep) {
  // 创建副本，防范在 cleanup 和 track 时的 Set 死循环
  const effectsToRun = new Set(dep);
  effectsToRun.forEach(effect => {
    // 规避 self-triggering（如 state.count++）导致的栈溢出
    if (effect !== activeEffect) {
      if (effect.scheduler) {
        effect.scheduler(); // 移交控制权
      } else {
        effect(); // 同步执行
      }
    }
  });
}
```

### 2. 📄 `reactive.js`（对象代理）
提供 `reactive` 的拦截与多层惰性代理逻辑。

```javascript
import { track, trigger } from './effect.js';

export function isObject(val) {
  return val !== null && typeof val === 'object';
}

// 代理标识，用以识别一个对象是否已经是 Proxy
const IS_REACTIVE = '__v_isReactive';

// 缓存原始对象与代理对象的映射，避免重复创建代理
const reactiveMap = new WeakMap();

const mutableHandlers = {
  get(target, key, receiver) {
    if (key === IS_REACTIVE) {
      return true;
    }

    track(target, key); // 依赖收集

    const res = Reflect.get(target, key, receiver);

    // 深度响应式：惰性代理，只有读取到深层对象时才包装
    if (isObject(res)) {
      return reactive(res);
    }

    return res;
  },

  set(target, key, value, receiver) {
    const oldValue = target[key];
    const result = Reflect.set(target, key, value, receiver);

    // 只有值确实改变时才触发更新
    if (oldValue !== value) {
      trigger(target, key);
    }
    return result;
  }
};

export function reactive(target) {
  if (!isObject(target)) {
    return target;
  }

  // 边界 2：如果传入的对象已经是代理对象，直接返回
  if (target[IS_REACTIVE]) {
    return target;
  }

  // 边界 1：如果已有缓存，直接复用
  const existingProxy = reactiveMap.get(target);
  if (existingProxy) {
    return existingProxy;
  }

  const proxy = new Proxy(target, mutableHandlers);
  reactiveMap.set(target, proxy); // 记录缓存
  return proxy;
}
```

### 3. 📄 `ref.js`（任意值代理与工具方法）
利用 Class 的 `getter/setter` 拦截单个值的存取，并实现相关的辅助转化函数。

```javascript
import { activeEffect, trackEffects, triggerEffects } from './effect.js';
import { reactive, isObject } from './reactive.js';

function hasChanged(value, oldValue) {
  return !Object.is(value, oldValue);
}

function toReactive(value) {
  return isObject(value) ? reactive(value) : value;
}

class RefImpl {
  constructor(value) {
    this._rawValue = value;
    this._value = toReactive(value); // 处理对象包裹情况
    this.dep = new Set();            // ref 直接内部持有独立的 dep Set
    this.__v_isRef = true;           // 标志是 ref
  }

  get value() {
    if (activeEffect) {
      trackEffects(this.dep);
    }
    return this._value;
  }

  set value(newVal) {
    if (hasChanged(newVal, this._rawValue)) {
      this._rawValue = newVal;
      this._value = toReactive(newVal);
      triggerEffects(this.dep);
    }
  }
}

export function ref(value) {
  return new RefImpl(value);
}

export function isRef(r) {
  return !!(r && r.__v_isRef === true);
}

export function unref(r) {
  return isRef(r) ? r.value : r;
}

class ObjectRefImpl {
  constructor(object, key) {
    this._object = object;
    this._key = key;
    this.__v_isRef = true;
  }

  get value() {
    return this._object[this._key]; // 读取时会间接触发 reactive 的 track
  }

  set value(newVal) {
    this._object[this._key] = newVal; // 写入时会间接触发 reactive 的 trigger
  }
}

export function toRef(object, key) {
  if (isRef(object[key])) {
    return object[key];
  }
  return new ObjectRefImpl(object, key);
}

export function toRefs(object) {
  const res = Array.isArray(object) ? new Array(object.length) : {};
  for (const key in object) {
    res[key] = toRef(object, key);
  }
  return res;
}
```

### 4. 📄 `computed.js`（计算属性）
结合 `lazy` 的配置和 `scheduler` 逻辑，处理计算属性的缓存和双向触发依赖。

```javascript
import { effect, trackEffects, triggerEffects, activeEffect } from './effect.js';

class ComputedRefImpl {
  constructor(getter) {
    this._dirty = true;
    this._value = undefined;
    this.dep = new Set(); // 外部依赖该计算属性的 effects 集合
    this.__v_isRef = true;

    // 创建底层的 lazy effect，配合 scheduler 挂载
    this.effect = effect(getter, {
      lazy: true,
      scheduler: () => {
        if (!this._dirty) {
          this._dirty = true; // 解开缓存锁
          triggerEffects(this.dep); // 触发收集的外部更新
        }
      }
    });
  }

  get value() {
    if (activeEffect) {
      trackEffects(this.dep); // 收集外部依赖当前 computed 的 effect
    }

    if (this._dirty) {
      this._dirty = false;
      this._value = this.effect(); // 执行计算，并缓存结果
    }
    return this._value;
  }
}

export function computed(getter) {
  return new ComputedRefImpl(getter);
}
```

### 第一步：为核心方法注入追踪日志

为了不破坏我们之前写好的代码结构，我们只需要在 `get`、`set`、`track`、`trigger` 等关卡中增加带有统一前缀的 `console.log`。

#### 1. 📄 在 `effect.js` 中增加日志追踪

```javascript
// effect.js
export let activeEffect = null;
export const targetMap = new WeakMap();

export function effect(fn, options = {}) {
  const effectFn = () => {
    let lastEffect = activeEffect;
    try {
      console.log(`\n🔮 [Effect 周期] 准备执行副作用。清理旧依赖...`);
      cleanupEffect(effectFn);
      
      activeEffect = effectFn;
      console.log(`🔮 [Effect 周期] activeEffect 标记为: [effectFn]`);
      
      return fn(); 
    } finally {
      activeEffect = lastEffect;
      console.log(`🔮 [Effect 周期] 副作用执行结束，恢复 activeEffect 指针。\n`);
    }
  };

  effectFn.deps = [];
  effectFn.options = options;
  effectFn.scheduler = options.scheduler;

  if (!options.lazy) {
    console.log(`🔮 [Effect 初始化] 创建并立即执行副作用...`);
    effectFn();
  }
  return effectFn;
}

export function track(target, key) {
  if (!activeEffect) {
    console.log(`   ⏭️ [track 跳过] 当前外部无活跃 effect，跳过本次依赖收集`);
    return;
  }

  console.log(`   📥 [track 开始] 检测到活动 effect。开始为 { ${key} } 搜寻依赖池...`);
  
  let depsMap = targetMap.get(target);
  if (!depsMap) {
    depsMap = new Map();
    targetMap.set(target, depsMap);
  }

  let dep = depsMap.get(key);
  if (!dep) {
    dep = new Set();
    depsMap.set(key, dep);
  }

  trackEffects(dep);
}

export function trackEffects(dep) {
  if (!dep.has(activeEffect)) {
    dep.add(activeEffect);
    activeEffect.deps.push(dep);
    console.log(`   📥 [track 成功] 双向绑定成功：已将当前 [effectFn] 存入该属性的 dep 集合中。`);
  } else {
    console.log(`   📥 [track 忽略] 该 [effectFn] 已存在于依赖集合中，无需重复收集。`);
  }
}

export function trigger(target, key) {
  console.log(`\n⚡ [trigger 触发] 检测到属性 "${key}" 被修改，开始搜寻并调度关联的 effects...`);
  const depsMap = targetMap.get(target);
  if (!depsMap) return;

  const dep = depsMap.get(key);
  if (!dep) return;

  triggerEffects(dep);
}

export function triggerEffects(dep) {
  const effectsToRun = new Set(dep);
  console.log(`⚡ [trigger 执行] 找到 ${effectsToRun.size} 个关联的 effects 准备运行。`);
  effectsToRun.forEach(effect => {
    if (effect !== activeEffect) {
      if (effect.scheduler) {
        console.log(`⚡ [trigger 调度] 检测到 scheduler，交由调度器处理。`);
        effect.scheduler();
      } else {
        console.log(`⚡ [trigger 执行] 执行副作用函数...`);
        effect();
      }
    }
  });
}

function cleanupEffect(effectFn) {
  const { deps } = effectFn;
  if (deps.length) {
    console.log(`   🧹 [Cleanup] 清理该 effect 之前挂载的所有依赖 (共 ${deps.length} 个)`);
    for (let i = 0; i < deps.length; i++) {
      deps[i].delete(effectFn);
    }
    deps.length = 0;
  }
}
```

#### 💡 重点时序分析：
1.  **初始化阶段**：
    *   `effect` 启动，最先执行的是 `cleanup`（清理机制），由于是首次执行，关联依赖个数为 `0` [1]。
    *   接着将全局指针 `activeEffect` 指向当前正在运行的副作用。
    *   开始运行用户函数时，读取了 `state.count`，从而**首先触发了 `get` 拦截**。
    *   在 `get` 内部，检测到存在 `activeEffect`，于是调用 `track`，把指针记录到 `count` 的依赖集合中，至此 **“依赖收集” 生命周期完成**。
2.  **触发更新阶段**：
    *   当我们进行赋值操作时，**首先触发了 `set` 拦截**。
    *   `set` 发现值确实改变了，调用 `trigger`。
    *   `trigger` 去刚才创建的全局依赖表 `targetMap` 中寻找到 `count` 的 Set 集合，把里面存着的 `effectFn` 提出来，**“派发更新” 生命周期开启**。
    *   再次进入副作用周期：执行前先执行 `Cleanup`（这次成功清除了 1 个旧关联），接着临时标记 `activeEffect`，执行用户逻辑，并通过再次触发 `get` 重新、动态地收集最新的依赖关系 [1]。

通过添加这套详尽的日志追踪，不仅能帮助你直观地查验代码有没有按照预期的生命周期流转，也是日后你在面对复杂的响应式 Bug 或者是进行性能调优时，最直观且可靠的调试手段。
