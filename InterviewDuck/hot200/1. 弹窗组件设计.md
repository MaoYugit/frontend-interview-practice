在前端面试中，设计一个 Vue 3 弹窗（Modal/Dialog）组件是一个非常经典且能全面考察候选人组件设计能力的问题。一个优秀的弹窗组件设计，不仅要考虑**基础的 API 易用性**，还要兼顾 **Vue 3 新特性**、**边界处理**以及**函数式（命令式）调用**。

---

### 一、 核心设计目标

1. **易用性与灵活性**：支持声明式（`<MyModal v-model="visible">`）和命令式（`Modal.show({ ... })`）两种调用方式。
2. **定制性**：通过 Props、Slots 保证内容、头部、底部高度可定制。
3. **用户体验与边界处理**：解决滚动穿透（Body Scroll Lock）、层级冲突（z-index 覆盖）、键盘事件（Esc 退出）、嵌套弹窗等问题。
4. **Vue 3 新特性应用**：充分利用 `<Teleport>`、`<Transition>` 和组合式 API（Composition API）。

### 二、 基础结构设计（声明式组件）

在 Vue 3 中，我们首先设计基础的声明式组件。利用 `<Teleport>` 可以将弹窗直接渲染到 `body` 下，避免父级容器 `overflow: hidden` 或 `position: relative` 带来的层级（z-index）失效问题。

```vue
<!-- MyModal.vue -->
<template>
  <!-- 使用 Teleport 将弹窗挂载到 body -->
  <Teleport to="body">
    <Transition name="modal-fade">
      <!-- 遮罩层 -->
      <div v-if="modelValue" class="modal-overlay" @click="handleOverlayClick">
        <!-- 弹窗主体 -->
        <div class="modal-container" :style="{ width }" @click.stop>
          <!-- 头部 -->
          <header class="modal-header">
            <slot name="header">
              <span class="modal-title">{{ title }}</span>
            </slot>
            <button class="modal-close" @click="close">×</button>
          </header>

          <!-- 内容区 -->
          <main class="modal-body">
            <slot></slot>
          </main>

          <!-- 底部按钮区 -->
          <footer class="modal-footer" v-if="$slots.footer">
            <slot name="footer"></slot>
          </footer>
        </div>
      </div>
    </Transition>
  </Teleport>
</template>

<script setup>
import { watch, onBeforeUnmount } from "vue";

// 1. 定义 Props
const props = defineProps({
  modelValue: { type: Boolean, default: false }, // 控制显示隐藏 (Vue 3 默认双向绑定)
  title: { type: String, default: "提示" },
  width: { type: String, default: "500px" },
  closeOnClickOverlay: { type: Boolean, default: true }, // 是否点击遮罩关闭
  closeOnEsc: { type: Boolean, default: true }, // 是否支持 Esc 键关闭
});

// 2. 定义 Emits
const emit = defineEmits(["update:modelValue", "close", "open"]);

const close = () => {
  emit("update:modelValue", false);
  emit("close");
};

const handleOverlayClick = () => {
  if (props.closeOnClickOverlay) close();
};

// 3. 键盘 Esc 关闭逻辑
const handleKeyDown = (e) => {
  if (props.closeOnEsc && e.key === "Escape" && props.modelValue) {
    close();
  }
};

// 4. 监听状态：处理滚动穿透与事件绑定
watch(
  () => props.modelValue,
  (val) => {
    if (val) {
      document.body.style.overflow = "hidden"; // 锁死背景滚动
      window.addEventListener("keydown", handleKeyDown);
      emit("open");
    } else {
      document.body.style.overflow = ""; // 恢复背景滚动
      window.removeEventListener("keydown", handleKeyDown);
    }
  },
);

onBeforeUnmount(() => {
  document.body.style.overflow = "";
  window.removeEventListener("keydown", handleKeyDown);
});
</script>

<style scoped>
/* 编写过渡动画与基础样式 */
.modal-fade-enter-active,
.modal-fade-leave-active {
  transition: opacity 0.3s ease;
}
.modal-fade-enter-from,
.modal-fade-leave-to {
  opacity: 0;
}
/* 遮罩及容器样式省略... */
</style>
```

### 三、 进阶功能设计（面试加分项）

在基本功能实现后，可以向面试官阐述如何处理以下几个实际业务中的痛点：

#### 1. 滚动穿透（Scroll Lock）优化

简单的 `document.body.style.overflow = 'hidden'` 在某些移动端浏览器上可能会失效，或者会导致页面右侧滚动条消失引发的页面“抖动”。

- **改进方案**：可以计算滚动条的宽度，在锁死滚动时给 `body` 加一个 `padding-right`；或者在生产环境中使用成熟的第三方库如 `body-scroll-lock`。

#### 2. 层级管理（Dynamic z-index）

如果页面上同时弹出了多个弹窗（嵌套弹窗），如何保证新弹出的弹窗始终在最上面？

- **改进方案**：维护一个全局的 `zIndex` 计数器（例如从 2000 开始）。每次弹窗被打开时，动态递增该值，并赋值给当前弹窗的 `style.zIndex`。

#### 3. 焦点捕获（Focus Trap）与无障碍（A11y）

对于追求无障碍体验的应用，弹窗打开时，键盘的 `Tab` 键切换焦点应该只在弹窗内部循环，不应该漂移到弹窗背后的页面元素上。

- **改进方案**：在弹窗打开时，暂存当前聚焦的元素，并将焦点强制移入弹窗。在弹窗关闭时，将焦点归还给原元素。

### 四、 函数式 / 命令式调用设计（Imperative API）

在实际业务中，我们经常需要通过 JS 直接调用弹窗，例如 `Modal.confirm({ title: '警告', content: '确定删除吗？' })`。在 Vue 3 中，我们可以利用 `createVNode` 和 `render` 函数动态创建并渲染组件。

我们可以设计一个辅助函数来实现它：

```javascript
// MyModalService.js
import { createVNode, render } from "vue";
import MyModal from "./MyModal.vue";

export const showModal = (options) => {
  // 1. 创建一个挂载的容器 DOM
  const container = document.createElement("div");
  document.body.appendChild(container);

  // 2. 定义销毁逻辑
  const destroy = () => {
    render(null, container); // 卸载组件
    container.remove(); // 移除容器
  };

  // 3. 构建虚拟节点 (vnode)
  const vnode = createVNode(
    MyModal,
    {
      ...options,
      modelValue: true, // 默认直接显示
      // 监听组件内部派发的事件
      "onUpdate:modelValue": (val) => {
        if (!val) destroy();
      },
      onClose: () => {
        if (options.onClose) options.onClose();
        destroy();
      },
    },
    {
      // 通过插槽渲染主体内容
      default: () =>
        typeof options.content === "function"
          ? options.content()
          : options.content,
    },
  );

  // 4. 将 vnode 渲染到 container 中
  render(vnode, container);
};
```

**使用方式**：

```javascript
import { showModal } from "./MyModalService";

showModal({
  title: "系统提示",
  content: "这是一条通过 JS 调用的弹窗消息。",
  onClose: () => {
    console.log("弹窗关闭了");
  },
});
```
